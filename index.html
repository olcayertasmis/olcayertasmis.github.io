public sealed class GameBootstrapper
{
  public void Initialize()
  {
    // Fail-fast DI & Zero-GC EventBus
    ServiceLocator.Register<IEconomy>(new EconomyService());
    ServiceLocator.Register<IRemoteConfig>(new SatoriConfig());

    EventBus.Publish(new OnSystemsReady {
      TargetFps = 60,
      AllocBytes = 0
    });
  }
}
